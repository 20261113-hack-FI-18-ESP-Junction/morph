def cycle_selection():
    return Selection(kind="cycle_explanation", sessionIds=["session-1"], cycleId="cycle-1", sourceMode="simulation")


def valid_output(context):
    comparison = bool(context["compatibility"])
    selected_metrics = [context["metrics"][0]["metricId"]]
    if context["evidenceFlags"]["hasRecoveredCycle"]:
        selected_metrics.append(next(metric["metricId"] for metric in context["metrics"] if metric["key"] == "recoveredCycles" and metric["value"] > 0))
    output = {"contractVersion": "gemma-brief-v1", "answerState": "insufficient_evidence" if comparison else "supported",
            "headline": analysis_provider.cycle_headline(context) if not comparison else "Comparison evidence is limited",
            "interpretations": [{"text": "The final cycle outcome and retained attempt outcomes use distinct populations.",
                                 "factIds": [context["facts"][1]["factId"]], "metricIds": selected_metrics}],
            "hypotheses": [], "evidenceGapKeys": [context["knownGaps"][0]["key"]], "suggestedReviewTargetIds": []}
    required = sorted(analysis_provider.required_recovery_metrics(context))
    if required:
        output["interpretations"][0]["metricIds"] = required[:4]
        output["interpretations"].append({"text": "The retry count is recorded separately.",
                                          "factIds": [context["facts"][1]["factId"]], "metricIds": required[4:]})
    requirements = analysis_provider.cycle_citation_requirements(context)
    for index, item in enumerate(output["interpretations"]):
        witnesses = set(item["factIds"])
        if index == 0:
            witnesses.update(requirements["requiredFactIds"])
        for identifier in item["metricIds"]:
            witnesses.update(requirements["metricFactIds"].get(identifier, []))
        item["factIds"] = sorted(witnesses)
    return output


def response(context, output=None):
    return {"httpStatus": 200, "providerLatencyMs": 4.0, "response": {
        "modelVersion": MODEL, "responseId": "test-response", "usageMetadata": {"totalTokenCount": 20},
        "candidates": [{"finishReason": "STOP", "content": {"parts": [{"thought": True, "text": "Hidden internal thought"},
                        {"text": json.dumps(output if output is not None else valid_output(context))}]}}]}}


class FakeProvider:
    def __init__(self):
        self.calls = []

    def __call__(self, context, question):
        self.calls.append((copy.deepcopy(context), question))
        return response(context)

def request(service, request_id="analysis-1", selection=None, **changes):
    selection = selection or cycle_selection()
    context = service.context(selection)
    return AnalysisRequest(**{**selection.model_dump(), "requestId": request_id,
                             "expectedSelectionDigest": context["contentDigest"], **changes})


def finished(service, report_id):
    deadline = time.monotonic() + 5
    while time.monotonic() < deadline:
        report = service.get(report_id)
        if report["status"] in {"ready", "rejected", "failed", "interrupted"}:
            return report
        time.sleep(0.01)
    pytest.fail("Bounded fake analysis did not finish")


def test_g1_recovery_context_exact_metrics_and_audit(recovered, analyst):
    before = unchanged_snapshot(recovered)
    context = analyst.context(cycle_selection())
    metrics = {row["key"]: row for row in context["metrics"]}
    assert (metrics["endToEnd"]["numerator"], metrics["endToEnd"]["denominator"]) == (1, 1)
    assert (metrics["physicalExecution"]["numerator"], metrics["physicalExecution"]["denominator"]) == (1, 2)
    assert metrics["physicalExecution"]["includedIds"] == [record["id"] for record in recovered.list_records("attempt")]
    assert [metrics[key]["value"] for key in ("retries", "recoveredCycles", "firstAttemptSuccesses")] == [1, 1, 0]
    attempt_values = [f["value"] for f in context["facts"] if f["sourceRefs"][0]["recordType"] == "attempt"]
    assert any(v.get("ordinal") == 1 and v["outcome"] == "failed" for v in attempt_values)
    assert any(v.get("ordinal") == 2 and v["outcome"] == "successful" for v in attempt_values)
    assert any(f["value"].get("previousValue") == "unverified" and f["value"].get("newValue") == "correct" for f in context["facts"])
    assert any("Historical revision" in f["text"] for f in context["facts"])
    assert len(canonical(context).encode()) <= MAX_CONTEXT_BYTES
    assert context["contentDigest"] == analyst.context(cycle_selection())["contentDigest"]
    report = finished(analyst, analyst.submit(request(analyst))["reportId"])
    assert report["status"] == "ready"
    assert report["validation"]["accepted"] is True
    assert report["provenance"]["requestedModel"] == report["provenance"]["returnedModel"] == MODEL
    assert report["timings"]["providerLatencyMs"] == 4.0
    assert all(report["timestamps"].values())
    assert unchanged_snapshot(recovered) == before

def test_correction_stales_report_and_blocks_old_selection(recovered, analyst):
    body = request(analyst)
    ready = finished(analyst, analyst.submit(body)["reportId"])
    saved_bytes = analyst._path(ready["reportId"]).read_bytes()
    cycle = recovered.get("cycle", "cycle-1")
    recovered.correct(correction_request(entityType="cycle", entityId="cycle-1", expectedRevision=cycle["revision"],
                                          field="outcome", previousValue="correct", newValue="unverified"))

    assert analyst._path(ready["reportId"]).read_bytes() == saved_bytes

    assert exc.value.status_code == 409 and len(analyst.provider.calls) == 1

@pytest.mark.parametrize("mutation", ["duplicate_json", "markdown", "nonfinite", "truncated", "model", "multiple", "function", "empty"])
def test_transport_and_json_rejections(analyst, mutation):

    candidate = result["response"]["candidates"][0]
    if mutation == "duplicate_json": candidate["content"]["parts"] = [{"text": '{"contractVersion":"x","contractVersion":"y"}'}]
    if mutation == "markdown": candidate["content"]["parts"][1]["text"] = "```json\n{}\n```"
    if mutation == "nonfinite": candidate["content"]["parts"][1]["text"] = '{"value":NaN}'

    if mutation == "function": candidate["content"]["parts"] = [{"functionCall": {"name": "dispatch"}}]
    if mutation == "empty": candidate["content"]["parts"] = [{"text": ""}]
    with pytest.raises(ProviderError): validate_reply(result, context)

def test_worker_is_single_and_queue_is_bounded(recovered, tmp_path):
    entered, release = threading.Event(), threading.Event()
    calls = []
    def blocked(context, question):
        calls.append(question)
        entered.set()
        assert release.wait(5)
        return response(context)
    service = AnalysisService(recovered, tmp_path / "bounded-analysis", blocked)
    second = AnalysisService(recovered, tmp_path / "bounded-analysis", FakeProvider())
    try:
        first = service.submit(request(service))
        assert entered.wait(2)
        for ordinal in range(4):
            service.submit(request(service, "queued-" + str(ordinal), question="Question " + str(ordinal)))
        with pytest.raises(StoreError) as exc:
            service.submit(request(service, "queue-overflow"))
        assert exc.value.status_code == 429
        with pytest.raises(StoreError) as exc:
            second.submit(request(second, "second-worker"))
        assert exc.value.status_code == 503 and len(calls) == 1
        release.set()
        assert finished(service, first["reportId"])["status"] == "ready"
    finally:
        release.set()
        second.close()
        service.close()

@pytest.mark.parametrize("failure", ["redirect", "limit", "http", "deadline", "nonfinite", "overflow", "duplicate"])
def test_actual_transport_bounds_and_failure_request_evidence(analyst, tmp_path, monkeypatch, failure):
    credential = tmp_path / "fake-provider.env"
    credential.write_text("GEMINI_API_KEY=unit-test-only-key\n", encoding="utf-8")
    credential.chmod(0o600)
    provider = analysis_provider.GemmaProvider(credential)
    called = []
    async def transport(request):
        called.append(request)
        if failure == "redirect": return httpx.Response(302, headers={"location": "https://unexpected.invalid"}, json={})
        if failure == "limit": return httpx.Response(200, content=b"x" * (analysis_provider.MAX_RESPONSE_BYTES + 1))
        if failure == "http": return httpx.Response(401, json={"error": {"message": "unit-test-only-key"}})
        if failure == "nonfinite": return httpx.Response(200, content=b'{"usageMetadata":{"totalTokenCount":NaN}}')
        if failure == "overflow": return httpx.Response(200, content=b'{"usageMetadata":{"totalTokenCount":1e999}}')
        if failure == "duplicate": return httpx.Response(200, content=b'{"candidates":[],"candidates":[{}]}')
        await asyncio.sleep(0.1)
        return httpx.Response(200, json={})
    original_client = httpx.AsyncClient
    monkeypatch.setattr(analysis_provider.httpx, "AsyncClient", lambda **kwargs: original_client(transport=httpx.MockTransport(transport), **kwargs))
    if failure == "deadline": monkeypatch.setattr(analysis_provider, "PROVIDER_DEADLINE_SECONDS", 0.01)
    with pytest.raises(ProviderError) as exc:
        provider(analyst.context(cycle_selection()), "Explain the evidence")
    assert len(called) == 1
    assert exc.value.evidence["request"]["endpoint"].endswith(MODEL + ":generateContent")
    assert exc.value.evidence["providerLatencyMs"] > 0
    assert "unit-test-only-key" not in canonical(exc.value.evidence)
    assert exc.value.code == {"redirect": "provider_http_error", "limit": "response_too_large", "http": "provider_http_error", "deadline": "provider_timeout",
                              "nonfinite": "invalid_transport_json", "overflow": "invalid_transport_json", "duplicate": "invalid_transport_json"}[failure]

def test_shutdown_after_dequeue_does_not_start_a_new_provider_call(recovered, tmp_path, monkeypatch):
    service = AnalysisService(recovered, tmp_path / "shutdown-race", FakeProvider())
    dequeued, release, stopping = threading.Event(), threading.Event(), threading.Event()
    original_read, original_put = service._read_report, service.jobs.put
    def gated_read(path):
        if threading.current_thread() is service.worker:
            dequeued.set()
            assert release.wait(3)
        return original_read(path)
    def observed_put(item, *args, **kwargs):
        if item is None: stopping.set()
        return original_put(item, *args, **kwargs)
    monkeypatch.setattr(service, "_read_report", gated_read)
    monkeypatch.setattr(service.jobs, "put", observed_put)
    report = service.submit(request(service))
    assert dequeued.wait(2)
    closer = threading.Thread(target=service.close)
    closer.start()
    try:
        assert stopping.wait(2)
        release.set()
        closer.join(3)
        assert not closer.is_alive() and service.provider.calls == []
        result = service.get(report["reportId"])
        assert result["status"] == "interrupted"
        assert result["timestamps"]["providerRequestedAt"] is None
        assert result["timestamps"]["providerCompletedAt"] is None
    finally:
        release.set()
        closer.join(3)
        service.close()
